    <title>Grouped Data Graph</title>
    <script src="https://cdn.jsdelivr.net/npm/chart.js"></script>

<body>
    <h2>Year Wise Birth Rate</h2>
    <canvas id="myChart" width="800" height="400"></canvas>
    <script>
        // Parse data from Flask, which should be a JSON array of arrays.
        const dataFromServer = JSON.parse('{{ data | tojson | safe }}');
    
        // Prepare an object to group counts by year and gender.
        const groupedData = {};
    
        dataFromServer.forEach(item => {
            // Access tuple elements by index:
            const year = item[0];      // First element (year)
            const gender = item[1];    // Second element (gender)
            const count = item[2];     // Third element (count)
    
            // Initialize group if needed
            if (!groupedData[year]) {
                groupedData[year] = { Male: 0, Female: 0 };
            }
            // Assign count to the proper gender for that year.
            groupedData[year][gender] = count;
        });
    
        // Create arrays for labels and datasets
        const labels = Object.keys(groupedData).sort();  // Sorted list of years
        const maleData = labels.map(year => groupedData[year]['Male'] || 0);
        const femaleData = labels.map(year => groupedData[year]['Female'] || 0);
    
        // Create a bar chart using Chart.js
        const ctx = document.getElementById('myChart').getContext('2d');
        const myChart = new Chart(ctx, {
            type: 'bar',
            data: {
                labels: labels,
                datasets: [
                    {
                        label: 'Male',
                        data: maleData,
                        backgroundColor: 'rgba(54, 162, 235, 0.6)',
                        borderColor: 'rgba(54, 162, 235, 1)',
                        borderWidth: 1
                    },
                    {
                        label: 'Female',
                        data: femaleData,
                        backgroundColor: 'rgba(255, 99, 132, 0.6)',
                        borderColor: 'rgba(255, 99, 132, 1)',
                        borderWidth: 1
                    }
                ]
            },
            options: {
                scales: {
                    y: {
                        beginAtZero: true,
                        ticks: {
                            stepSize: 1
                        }
                    }
                },
                responsive: true,
                plugins: {
                    legend: {
                        position: 'top',
                    },
                    title: {
                        display: true,
                        text: 'Birth Year and Gender Distribution'
                    }
                }
            }
        });
    </script>    
</body>
